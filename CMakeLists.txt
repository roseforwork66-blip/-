cmake_minimum_required(VERSION 3.16)
project(smart_guide_cane LANGUAGES CXX)

add_library(guide_cane firmware/GuideCane/src/control.cpp firmware/GuideCane/src/lidar.cpp)
target_include_directories(guide_cane PUBLIC firmware/GuideCane/src)
target_compile_features(guide_cane PUBLIC cxx_std_17)
set_target_properties(guide_cane PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/W4 /permissive-)
    target_compile_options(guide_cane PRIVATE /W4 /permissive-)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
    target_compile_options(guide_cane PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(cane_demo_scenarios STATIC tools/demo_scenarios.cpp)
target_include_directories(cane_demo_scenarios PUBLIC tools)
target_link_libraries(cane_demo_scenarios PUBLIC guide_cane)

add_executable(cane_demo tools/demo.cpp)
target_link_libraries(cane_demo PRIVATE cane_demo_scenarios)

enable_testing()
add_executable(cane_tests tests/test_cane.cpp)
target_link_libraries(cane_tests PRIVATE guide_cane)
add_test(NAME cane_tests COMMAND cane_tests)

if(WIN32)
    add_executable(SmartGuideCane WIN32 windows/dashboard.cpp windows/dashboard.rc)
    target_include_directories(SmartGuideCane PRIVATE windows)
    target_link_libraries(SmartGuideCane PRIVATE cane_demo_scenarios user32 gdi32 comctl32)
    target_compile_definitions(SmartGuideCane PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(SmartGuideCane PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
    elseif(MINGW)
        target_link_options(SmartGuideCane PRIVATE -municode)
    endif()
    add_test(NAME windows_demo_smoke COMMAND SmartGuideCane --smoke-test)
endif()

option(SMART_GUIDE_CANE_LINUX_GUI "Build the Linux GTK desktop demo" OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND SMART_GUIDE_CANE_LINUX_GUI)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
    add_executable(smart-guide-cane linux/dashboard.cpp)
    target_link_libraries(smart-guide-cane PRIVATE cane_demo_scenarios PkgConfig::GTK3)
    find_program(XVFB_RUN xvfb-run)
    if(XVFB_RUN)
        add_test(NAME linux_demo_smoke COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:smart-guide-cane> --smoke-test)
    endif()
    include(linux/Packaging.cmake)
endif()
